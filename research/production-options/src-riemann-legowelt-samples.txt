_addEvent = function (o, t, f) {
	  if (o.addEventListener) o.addEventListener(t, f, false);
	  else if (o.attachEvent) o.attachEvent('on'+ t, f);
	  else o['on'+ t] = f;
};

	  if (o.removeEventListener) o.removeEventListener(t, f, false);

	  else o['on'+ t] = null;
};

        'transform':'transform'
    };
