### STEP 5 - render MIDI to audio offline
import time, numpy as np, soundfile as sf, mido
from pedalboard import load_plugin
p = load_plugin("./surge/Surge XT.vst3", initialization_timeout=30.0)
sr=48000
msgs=[]
t=0.0
for i,n in enumerate([36,43,48,55]):
    msgs.append(mido.Message('note_on', note=n, velocity=100, time=t))
    msgs.append(mido.Message('note_off', note=n, time=t+1.4))
    t+=1.5
t0=time.time()
audio = p(msgs, duration=7.0, sample_rate=sr, num_channels=2)
el=time.time()-t0
a=np.asarray(audio)
print("shape", a.shape, "dtype", a.dtype)
print("peak %.4f rms %.5f" % (float(np.abs(a).max()), float(np.sqrt((a**2).mean()))))
print("render wall time %.2fs for 7.0s audio -> %.1fx realtime" % (el, 7.0/el))
sf.write("surge_test.wav", a.T, sr)
# now change a parameter and re-render
prm=[k for k in p.parameters if 'cutoff' in k.lower()][:5]
print("cutoff-ish params:", prm)

### STEP 6 - parameter changes; the disabled-module gotcha
import numpy as np, mido
from pedalboard import load_plugin
p = load_plugin("./surge/Surge XT.vst3", initialization_timeout=30.0)
sr=48000
msgs=[mido.Message('note_on', note=36, velocity=110, time=0.0), mido.Message('note_off', note=36, time=2.0)]
def rms(a): return float(np.sqrt((np.asarray(a)**2).mean()))
print("filter type param:", p.a_filter_1_type if hasattr(p,'a_filter_1_type') else [k for k in p.parameters if 'filter_1' in k][:8])
base=rms(p(msgs, duration=3.0, sample_rate=sr))
for name,val in [('global_volume',-12.0),('a_osc_1_pitch',12.0)]:
    if name in p.parameters:
        old=getattr(p,name); setattr(p,name,val)
        r=rms(p(msgs,duration=3.0,sample_rate=sr))
        print(f"{name}: {old} -> {getattr(p,name)}  rms {base:.5f} -> {r:.5f} changed={abs(base-r)>1e-5}")
        setattr(p,name,old)
    else:
        print(name,"absent; candidates:",[k for k in p.parameters if name.split('_')[-1] in k][:6])

### STEP 8 - DawDreamer: Faust drum synthesis with NO external plugin, and the same VST3
import dawdreamer as daw, numpy as np, time
SR=48000
eng=daw.RenderEngine(SR,512)
f=eng.make_faust_processor("f")
f.set_dsp_string("""
import("stdfaust.lib");
sy = library("synths.lib");
gate = os.lf_pulsetrain(2.0, 0.05) > 0.5;
k = sy.kick(55, 0.2, 0.01, 0.5, 6.0, gate);
process = k <: _,_;
""")
print("faust compiled:", f.compile())
eng.load_graph([(f,[])])
t0=time.time(); eng.render(4.0); el=time.time()-t0
a=eng.get_audio()
print("faust sy.kick -> shape",a.shape,"peak %.3f"%float(np.abs(a).max()),"| %.3fs wall for 4.0s (%.0fx RT)"%(el,4.0/el))
p=eng.make_plugin_processor("surge","./surge/Surge XT.vst3")
print("plugin loaded:", p.get_name(), "| params:", p.get_plugin_parameter_size())
p.add_midi_note(36,110,0.0,1.0); p.add_midi_note(48,100,1.0,1.0)
eng.load_graph([(p,[])])
t0=time.time(); eng.render(3.0); el=time.time()-t0
b=eng.get_audio()
print("Surge via DawDreamer -> shape",b.shape,"peak %.3f"%float(np.abs(b).max()),"| %.3fs wall for 3.0s (%.0fx RT)"%(el,3.0/el))
